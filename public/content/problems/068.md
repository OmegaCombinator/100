```acorn
from list.list_sum import partial
from nat import Nat
from real.real_base import Real
from real.real_series import partial_suc
from semiring.from_nat import from_nat

define arith(a: Real, d: Real) -> Nat -> Real {
    function(k: Nat) { a + from_nat[Real](k) * d }
}

theorem arith_partial_suc(a: Real, d: Real, n: Nat) {
    partial[Real](arith(a, d), n.suc) = partial[Real](arith(a, d), n) + arith(a, d)(n)
} by {
    partial_suc(arith(a, d), n)
}
```

## Interpretation

This confirms the finite partial-sum recurrence for arithmetic sequences over `Real`. The next step is the closed form for finite sums of linear functions.
